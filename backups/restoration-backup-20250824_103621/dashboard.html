' + (profileNames[profile] || profile) + '
' ).join(''); } // Função para trocar perfil function switchProfile(profile) { const profileRoutes = { 'EMPLOYER': 'dashboard-employer.html', 'EMPLOYEE': 'dashboard-employee.html', 'FAMILY': 'dashboard-family.html', 'ADMIN': 'dashboard-admin.html' }; const route = profileRoutes[profile] || 'dashboard.html'; window.location.href = route; }
